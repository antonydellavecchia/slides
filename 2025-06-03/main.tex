\begin{frame}[fragile]{Algebraic Shifting}
  \begin{definition}
    $C \in \binom{[n]}{k}$ is \emph{shifted} if for any $\sigma \in C$,
    $\RepVert{\sigma}{i}{j} \in C$ for $i \in \sigma$ and $ j < i$.
  \end{definition}

  \begin{theorem}[Kalai '84, Bj\"orner-Kalai '88]
    Given $K$ a simplicial complex and a field $\FF$, there exists a shifted simplicial complex
    $\Delta^{\FF}(K)$ with the same $f$-vector and Betti numbers as $K$.
  \end{theorem}

\begin{frame}[fragile]{Exterior Algebraic Shifting}
  \begin{definition}
    Let $\EE \supset \FF$ be a field extension and let $g \in \GL(n, \EE)$.
    \begin{itemize}
      \item The $k$th \emph{compound matrix} $g^{\wedge k}$ is the $\binom{n}{k} \times \binom{n}{k}$-matrix
        of $k$-minors of $g$.
        \item $g^{\wedge K_k}$ is the $\abs{K_k} \times \binom{n}{k}$-submatrix of $g^{\wedge k}$ with rows indexed by $K_k$.
    \end{itemize}
    \end{definition}
\begin{definition}\label{def:shift}
  The \emph{partial (exterior) shift} of $K$ by $g$ is
  \begin{equation*}
    \Delta^{\FF}_g(K) \coloneqq \bigcup_{k} \Set[\big]{ \sigma \in \tbinom{[n]}{k}; g^{\wedge K_k}_{*\sigma} \notin \Span_{\EE}[\big]{(g^{\wedge K_k}_{*\rho})_{\rho \lex< \sigma}} }.
  \end{equation*}
\end{definition}
\pause
\begin{example}
  For $g = (x_{ij})_{1\leq i, j\leq 3}$ and $K = \{ \{1, 2\}, \{2, 3\}\}$.
  \begin{center}
    $g^{\wedge K_2} = \begin{psmallmatrix} x_{11}x_{22} - x_{12}x_{21} & x_{11}x_{23} - x_{13}x_{21} &  x_{12}x_{23} - x_{22}x_{13} \\
      x_{21}x_{32} - x_{22}x_{31} & x_{21}x_{33} - x_{23}x_{31} &  x_{22}x_{33} - x_{23}x_{32}
    \end{psmallmatrix}$,
  \end{center}
  $\Delta(K) = \{\{1, 2\}, \{1, 3\}\}$
  \end{example}
  %mention monte carlo
\end{frame}

\begin{frame}[fragile]{Algebraic Exterior Shifting}
  \textbf{Generic Initial Ideals}\\
  For $\sigma = \{\sigma_1 < \ldots < \sigma_k\}$ set $e_{\sigma} = e_{\sigma_1} \wedge \ldots \wedge e_{\sigma_k}$ 
  \[
    I_K \coloneqq (e_\sigma\colon \sigma \notin K) \subset \bigwedge \FF^n 
  \]

  The exterior face ring is,
  \[
    \FF[K] \coloneq {\bigwedge \FF^n}\Big/{ I_k},
  \]

  and

  \[
    \FF[\Delta(K)] = {\bigwedge \FF^n}\Big/{ \text{gin}(I_k)}
  \]
  See [Herzog-Tarai '99] Stable Properties of algebraic shifting.
\end{frame}

\begin{frame}[fragile]{Partial Algebraic Shifting }
  Consider now $\Set{\frR(w) \in \GL(n, \EE); w \in S_n}$.\\
  Let $c_n \coloneqq (1 \ldots n) \in S_n$.

  \begin{theorem}[DV-Joswig-Lenzen '24+]
    For any simplical complex $K$ and $w \in S_n$ such that $c_n \leq w$ in the weak Bruhat order we have the following,
    \begin{enumerate}
    \item $\Delta_{\frR(w)}(K)$ is homotopy equivalent to a wedge of spheres.
    \item $\Delta_{\frR(w)}(K)$ has the same Betti numbers as $K$.
    \end{enumerate}

  \end{theorem}
\end{frame}

\begin{frame}[fragile]{Matrix decompositions }
  \begin{lemma}
    $\Delta_{gb}(K) = \Delta_g(K)$ for any upper triangular $b$ and any simplicial complex $K$.
  \end{lemma}
  \vspace{8mm}

  \textbf{LU decomposition}
    For $g \in \GL(n, \EE)$ with non zero leading principal minors, there exists
    $l$ lower triangular, $u$ upper triangular such that

    \[
      g = lu
    \]
    \vspace{3mm}

    \textbf{Bruhat Decomposition}
    Let $B \subset \GL(n, \EE)$ be the subgroup of upper triangular matrices, 
    \begin{equation*}
      \GL(n, \EE) \ = \ \smashoperator{\coprod_{w \in S_n}} BwB \enspace .
    \end{equation*}
  \end{frame}

\begin{frame}[fragile]{ Finding the Bruhat cell of $g$}
  \[
    w_0 = (1 \enspace n)(2 \enspace n - 1)\ldots(\floor{\frac{n}{2}} \ceil{\frac{n}{2} + 1})
  \]

  If $g$ is generic, then so is $w_0g$, hence
  \[
    \Delta_{w_0g}(K) = \Delta_{g}(K) = \Delta(K).
  \]

  For the same reason $w_0g$ there exist some lower triangular $l$, and upper triangular $u', u$ such that,
    \begin{align*}
      w_0g &= lu = w_0u'w_0u\\
      g &= u'w_0u
    \end{align*}
    $g \in Bw_0B$
\end{frame}

\begin{frame}[fragile]{Inversions}
  \begin{definition}
    For $w \in S_n$, let
    \[
      \inv w \ \coloneqq \ \Set{(i,j); i < j, i \cdot w > j \cdot w} \enspace 
    \]
    % inversion table Stanley enumerative combinatorics
    \[
            \ell(w) \ \coloneqq \ \abs{\inv w}
          \]

          \[
            w_0 = \text{argmax}_{w \in S_n} \ell(w)
          \]
            \pause
    \begin{example}
    Take $w = (2 \enspace 3)$, $\inv w = \{ (1, 2), (1, 3)\}$.
    \[
      \begin{psmallmatrix}
	1 & x & y \\
          & 1 & z \\
          &   & 1
      \end{psmallmatrix}
      \begin{psmallmatrix}
	1 &   &   \\
          &   & 1 \\
          & 1 &
      \end{psmallmatrix}
      \ = \
      \begin{psmallmatrix}
	1 & x & y \\
          & 1 & z \\
          &   & 1
      \end{psmallmatrix}
      \begin{psmallmatrix}
	1 &   &   \\
          &   & 1 \\
          & 1 &
      \end{psmallmatrix} 
      \ = \
      \begin{psmallmatrix}
	1 & 0 & 0 \\
          & 1 & z \\
          &   & 1
      \end{psmallmatrix} 
      \begin{psmallmatrix}
	1 &   &   \\
          &   & 1 \\
          & 1 &
      \end{psmallmatrix}
      \begin{psmallmatrix}
	1 & y & x \\
          & 1 & 0 \\
          &   & 1
      \end{psmallmatrix} \enspace.
      \]
  \end{example}
\pause
          \textbf{Minimal representatives}\\
    \begin{equation*}
      \frU(w) \coloneqq
      \begin{psmallmatrix}
        1 & u_{12} & \cdots & u_{1n}    \\
          & \ddots & \ddots & \vdots    \\
          &        & 1      & u_{n-1,n} \\
          &        &        & 1
      \end{psmallmatrix}
      \quad \text{with} \quad
      u_{ij} = \begin{cases*}
        x_{ij} & if $(i,j) \in \inv w$,\\
        0 & otherwise
      \end{cases*}\enspace.
    \end{equation*}
    \vspace{3mm}
    We call $\frR(w) \coloneqq \frU(w)w$ the \emph{Rothe matrix}.
  \end{definition}
\end{frame}

\begin{frame}[fragile]{Weak Bruhat order}
  \begin{definition}[Bj\"orner-Brenti '05]
    The weak Bruhat right order is a partial ordering on $S_n$,
    \[
      w = uv \geq u
    \]
    if $\ell(w) = \ell(u) + \ell(v)$.
  \end{definition}

  \begin{lemma}
    For $u, v \in S_n$ with $uv \geq u$ if and only if
    \begin{align*}
      \inv uv &= \inv u \cup (\inv v) \cdot u^{-1} \\
      \emptyset &= \inv u \cap \inv v
    \end{align*}
  \end{lemma}
\end{frame}


\begin{frame}[fragile]{Rothe Matrix of the n cycle}
  Consider $c_n = (1 \enspace 2 \dots n)$,

  \begin{itemize}
  \item For $ 1 \leq i < j < n$, $i \cdot c_n < j \cdot c_n$
  \item $\inv c_n = \{ (i, n) : 1 \leq i < n \}$
  \end{itemize}

  \[
    \frR(c_n) = \begin{psmallmatrix}
      x_{1,n}   & 1 &        &   \\
      \vdots    &   & \ddots &   \\
      x_{n-1,n} &   &        & 1 \\
      1         & 0 & \cdots & 0
    \end{psmallmatrix}.
  \]

  For $w \geq c_n$
  \begin{itemize}
  \item $\inv c_n \subset \inv w$
  \item $\frR(c_n)$ and $\frR(w)$ have the same first column.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{ Near cones}
  \begin{definition}
    A simplicial complex $K$ is a \emph{near cone} if for every $\sigma \in K$ with $1 \notin \sigma$ we have $\RepVert{\sigma}{i}{1} \in K$
    for each $i \in \sigma$.
  \end{definition}

  \begin{lemma}[Bj\"orner-Kalai '88]
    Let $K$ be a near cone. 
    Then $K$ is homotopy equivalent to a wedge of spheres and $\beta_k(K) = \abs{\Set{\sigma \in K^k ; \sigma \cup \{1\} \notin K}}$.
  \end{lemma}

\end{frame}

\begin{frame}[fragile]{Permutation Lemma}
  \begin{lemma}[Bj\"orner-Kalai '88]
    Let $g \in \GL(n, \EE)$ be generic and suppose for $\tau \in \binom{[n]}{k}$ there exists    $\gamma \in \EE$ such that,
    \[
      g^{\wedge K_k}_{\tau} = \sum_{\sigma \in K_k}\gamma_{\sigma} g^{\wedge K_k}_{\sigma}.
    \]
    If $\pi \in S_n$, then there exists $\gamma' \in \EE$ such that
    \[
      g^{\wedge K_k}_{\tau \cdot \pi} = \sum_{\sigma \in K_k}\gamma'_{\sigma \cdot \pi} g^{\wedge K_k}_{\sigma \cdot \pi}
    \]
  \end{lemma}

\end{frame}

\begin{frame}[fragile]{Shifting to near cones}
  \begin{lemma}[DV-Joswig-Lenzen '24+]
    Let $K$ be a simplicial complex on $n$ vertices.
    Further, let $g\in\GL(n,\EE)$ be a matrix whose entries in the first column are algebraically independent over $\FF$ and the other entries of $g$.
    Then $\Delta_g(K)$ is a near cone.
  \end{lemma}

  \textbf{Proof sketch} \\
  Let $\pi = (1 \enspace i)$ and suppose $\tau \notin K$,
    \[
      g^{\wedge K_k}_{\tau} = \sum_{\sigma < \tau}\gamma_{\sigma} g^{\wedge K_k}_{\sigma}.
    \]

    Then there exists $\gamma' \in \EE$ such that,
    \[
      g^{\wedge K_k}_{\tau \cdot \pi} = \sum_{\sigma < \tau \cdot \pi}\gamma'_{\sigma } g^{\wedge K_k}_{\sigma}
    \]

\end{frame}

\begin{frame}[fragile]{Preserving Betti Numbers}
  \begin{lemma}[DV-Joswig-Lenzen '24+]
    Let $v$, $w \in S_n$ with $\ell(vw) = \ell(v)+\ell(w)$ then  $\frR(v w)^{K_k}$ and
    $(\frR(v) \frR(w)^v)^{\wedge K_k}$.
    have the same column matroid.
  \end{lemma}

  \textbf{Proof sketch} of shifting by $w \geq c_n$ preserves Betti numbers \\
  It is enough to show,
  \[
    \abs{\Set{\sigma \in \Delta_{\frR(w_0)}(K^k); 1 \in \sigma}} = \abs{\Set{\sigma \in \Delta_{\frR(w)}(K^k); 1 \in \sigma}}.
  \]

  Use lemma and do some algebra in the subscripts of $\Delta_{\frR(w)}$ and $\Delta_{\frR(w_0)}$
\end{frame}
