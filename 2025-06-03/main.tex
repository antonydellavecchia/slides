\begin{frame}[fragile]{Algebraic Shifting}
  \begin{definition}
    $C \in \binom{[n]}{k}$ is \emph{shifted} if for any $\sigma \in C$,
    $\RepVert{\sigma}{i}{j} \in C$ for $i \in \sigma$ and $ j < i$.
  \end{definition}

  \begin{theorem}[Kalai, Bj\"orner-Kalai]
    Given $K$ a simplicial complex and a field $\FF$, there exists a shifted simplicial complex
    $\Delta^{\FF}(K)$ with the same $f$-vector and betti numbers as $K$.
  \end{theorem}

\begin{frame}[fragile]{Exterior Algebraic Shifting}
  \begin{definition}
    Let $\EE \supset \FF$ be a field extension and let $g \in \GL(\EE, n)$.
    \begin{itemize}
      \item The $k$th \emph{compound matrix} $g^{\wedge k}$ is the $\binom{n}{k} \times \binom{n}{k}$-matrix
        of $k$-minors of $g$.
        \item $g^{\wedge K_k}$ is the $\abs{K_k} \times \binom{n}{k}$-submatrix of $g^{\wedge k}$ with rows indexed by $K_k$.
    \end{itemize}
    \end{definition}
\begin{definition}\label{def:shift}
  The \emph{partial (exterior) shift} of $K$ by $g$ is
  \begin{equation*}
    \Delta^{\FF}_g(K) \coloneqq \bigcup_{k \leq \text{dim}(K)} \Set[\big]{ \sigma \in \tbinom{[n]}{k}; g^{\wedge K_k}_{*\sigma} \notin \Span_{\EE}[\big]{(g^{\wedge K_k}_{*\rho})_{\rho \lex< \sigma}} }.
  \end{equation*}
\end{definition}
For $g = (x_{ij})_{1\leq i, j\leq 3}$ and $K = \{ \{1, 2\}, \{2, 3\}\}$.
\begin{center}
  $g^{\wedge 2} = \begin{psmallmatrix} x_{11}x_{22} - x_{12}x_{21} & x_{11}x_{23} - x_{13}x_{21} &  x_{12}x_{23} - x_{22}x_{13} \\
    x_{21}x_{32} - x_{22}x_{31} & x_{21}x_{33} - x_{23}x_{31} &  x_{22}x_{33} - x_{23}x_{32}
  \end{psmallmatrix}$,
  \end{center}
  $\Delta(K) = \{\{1, 2\}, \{1, 3\}\}$
  mention monte carlo
\end{frame}

\begin{frame}[fragile]{Partial Algebraic Shifting }
  Consider now $\Set{\frR(w) \in \GL(\EE, n); w \in S_n}$.\\
  Let $c_n \coloneqq (1 \ldots n) \in S_n$.

  \begin{theorem}[DV-Joswig-Lenzen]
    For any simplical complex $K$ and $w \in S_n$ such that $c_n \leq w$ in the weak Bruhat order we have the following,
    \begin{enumerate}
    \item $\Delta_{\frR(w)}(K)$ is a homotopy equivalent to a wedge of spheres.
    \item $\Delta_{\frR(w)}(K)$ has the same Betti numbers as $K$.
    \end{enumerate}

  \end{theorem}
\end{frame}

\begin{frame}[fragile]{LU decomposition }
  \begin{lemma}
    $\Delta_{gb}(K) = \Delta_g(K)$ for any upper triangular $b$ and any simplicial complex $K$.
  \end{lemma}
  \vspace{8mm}

  \textbf{LU decomposition}
    For $g \in \GL(n, \EE)$ with non zero leading principal minors, there exists
    $l$ lower triangular, $u$ upper triangular such that

    \[
      g = lu
    \]
    \vspace{8mm}

    For generic $g \in \GL(n, \EE)$ , $\Delta_l(K) = \Delta_g(K)$.
    
    I.e. we can use $l$ with $\frac{1}{2}n(n-1)$ transcendentals.
  \end{frame}

\begin{frame}[fragile]{Inversions and Rothe matrix}
  \begin{definition}
    For $w \in S_n$, let
    \[
      \inv w \ \coloneqq \ \Set{(i,j); i < j, i \cdot w > j \cdot w} \enspace 
    \]

    \[
            \ell(w) \ \coloneqq \ \abs{\inv w}.
    \]
    % inversion table Stanley enumerative combinatorics

    \begin{equation*}
      \frU(w) \coloneqq
      \begin{psmallmatrix}
        1 & u_{12} & \cdots & u_{1n}    \\
          & \ddots & \ddots & \vdots    \\
          &        & 1      & u_{n-1,n} \\
          &        &        & 1
      \end{psmallmatrix}
      \quad \text{with} \quad
      u_{ij} = \begin{cases*}
        x_{ij} & if $(i,j) \in \inv w$,\\
        0 & otherwise,
      \end{cases*}
    \end{equation*}
    and let $\frR(w) \coloneqq \frU(w)w$.
  \end{definition}

  \begin{example}
    Take $w = (2 \enspace 3)$, $\inv w = \{ (1, 2), (1, 3)\}$.

    \[
      \begin{psmallmatrix}
	1 & x & y \\
          & 1 & z \\
          &   & 1
      \end{psmallmatrix}
      \begin{psmallmatrix}
	1 &   &   \\
          &   & 1 \\
          & 1 &
      \end{psmallmatrix}
      \ = \
      \begin{psmallmatrix}
	1 & y & x \\
	0  & z & 1 \\
	0  & 1 & 0
      \end{psmallmatrix}
      \ = \
      \begin{psmallmatrix}
	1 & 0 & 0 \\
          & 1 & z \\
          &   & 1
      \end{psmallmatrix}
      \begin{psmallmatrix}
	1 &   &   \\
          &   & 1 \\
          & 1 &
      \end{psmallmatrix}
      \begin{psmallmatrix}
	1 & y & x \\
          & 1 & 0 \\
          &   & 1
      \end{psmallmatrix} \enspace.
    \]
  \end{example}
\end{frame}

\begin{frame}[fragile]{ Near cones}
  \begin{definition}
    A simplicial complex $K$ is a \emph{near cone} if for every $\sigma \in K$ with $1 \notin \sigma$ we have $\RepVert{\sigma}{i}{1} \in K$
    for each $i \in \sigma$.
  \end{definition}

  \begin{lemma}[Bj\"orner-Kalai 88]
    Let $K$ be a near cone. 
    Then $K$ is homotopy equivalent to a wedge of spheres and $\beta_k(K) = \abs{\Set{\sigma \in K^k ; \sigma \cup \{1\} \notin K}}$.
  \end{lemma}

\begin{frame}[fragile]{Shifting to near cones}
  \begin{lemma}[DV-Joswig-Lenzen]
    Let $K$ be a simplicial complex on $n$ vertices.
    Further, let $g\in\GL(n,\EE)$ be a matrix whose entries in the first column are algebraically independent over $\FF$ and the other entries of $g$.
    Then $\Delta_g(K)$ is a near cone.
  \end{lemma}

  \begin{example}
	The $n$-cycle $c_n \coloneqq (1\ 2\ \ldots\ n)$ gives rise to the matrix
	\[
	\frR(c_n) = \begin{psmallmatrix}
		x_{1,n}   & 1 &        &   \\
		\vdots    &   & \ddots &   \\
		x_{n-1,n} &   &        & 1 \\
		1         & 0 & \cdots & 0
	\end{psmallmatrix}.
      \]
      \begin{align*}
        \Delta^{\QQ}_{\frR(c_6)}(\RP^2_6) &=\Delta\{123, 124, 125, 126, 134, 135, 136, 145, 146, 156\} \\
        &=\Delta^{\QQ}(\RP^2_6)
      \end{align*}
\end{example}


\end{frame}

\begin{frame}[fragile]{Shifting to near cones}
  \begin{lemma}[Bj\"orner-Kalai]
    Let $g \in \GL(n, \EE)$ be generic and suppose for $\tau \in \binom{[n]}{k}$ there exists    $\gamma \in \EE$ such that,
    \[
      g^{\wedge K_k}_{\tau} = \sum_{\sigma \in K_k}\gamma_{\sigma} g^{\wedge K_k}_{\sigma}.
    \]
    If $w \in S_n$, then there exists $\gamma' \in \EE$ such that
    \[
      g^{\wedge K_k}_{\tau \cdot w} = \sum_{\sigma \in K_k}\gamma'_{\sigma \cdot w} g^{\wedge K_k}_{\sigma \cdot w}
    \]
  \end{lemma}

  spell out connection with generic first column shifting to near cone
  
\end{frame}

\begin{frame}[fragile]{Proof of results}
  \begin{theorem}[DV-Joswig-Lenzen]
    For $K$ a simplicial complex, $v$, $w \in S_n$ with $\ell(vw) = \ell(v)+\ell(w)$,
    \[
      \Delta_{\frR(v w)}(K) = \Delta_{\frR(v) \: \frR(w)^v}(K)
    \]
  \end{theorem}
