\begin{frame}[fragile]{Simplicial Complexes}
  \begin{definition}
    \begin{itemize}
    \item An \emph{abstract simplicial complex} $K$, is a collection of non empty sets of $[n] := \{1, 2,\ldots, n\}$,
      such that for any $\tau, \sigma \subset [n]$, if $\tau \subset  \sigma \in K$ then $\tau \in K$.
    \item We call $\sigma \in K$ a \emph{$k$-simplex} or a \emph{face} of dimension $k$ if $\abs{\sigma} = k + 1$.
    \item The \emph{$f$-vector} counts the number of $k$-faces for each $k$.
    \end{itemize}
  \end{definition}

\begin{frame}[fragile]{Simplicial Homology}
  \begin{definition}
    Given a field $\FF$, and a simplicial complex $K$ we define the
    $k$th simplicial homology to be the vector space
    \[
      H_k^{\FF}(K) \coloneq {k \text{-cycles}}\Big/{ k + 1 \text{-boundaries} }.
    \]
    The $k$th \emph{Betti number} of $K$ is the dimension of $H_k^{\FF}(K)$.
  \end{definition}

\begin{frame}[fragile]{Algebraic Shifting}
  \begin{definition}
    A \emph{$k$-uniform hypergraph} is collection $C \in \binom{[n]}{k}$.
    $C$ is \emph{shifted} if for any $\sigma \in C$,
    $\RepVert{\sigma}{i}{j} \in C$ for $i \in \sigma$ and $ j < i$.
  \end{definition}

\begin{frame}[fragile]{Compound Matrices}
  \begin{definition}
    Let $\EE \supset \FF$ be a field extension and let $g \in \GL(n, \EE)$.
    \begin{itemize}
      \item The $k$th \emph{compound matrix} $g^{\wedge k}$ is the $\binom{n}{k} \times \binom{n}{k}$-matrix
        of $k$-minors of $g$.
        \item $g^{\wedge K_k}$ is the $\abs{K_k} \times \binom{n}{k}$-submatrix of $g^{\wedge k}$ with rows indexed by $K_k$.
    \end{itemize}
  \end{definition}
  Let $\sigma_i \in K_k$, $\abs{K_k} = m$ and $\tau_i \in \binom{[n]}{k}$ such that
  \begin{center}
    $\sigma_1 <_{\text{lex}} \sigma_2  <_{\text{lex}} \dots <_{\text{lex}} \sigma_m$ and   $\tau_1  <_{\text{lex}} \tau_2 <_{\text{lex}} \dots  <_{\text{lex}} \tau_{\binom{n}{k}}$
  \end{center}

    \[
      g^{\wedge K_k} \ = \ \begin{psmallmatrix} \det g_{\sigma_1 \tau_1} & \det g_{\sigma_1 \tau_2} & \dots &  \det g_{\sigma_m \tau_{\binom{n}{k}}} \\
        \vdots & \vdots & \vdots & \vdots \\
        \det g_{\sigma_m \tau_1} & \det g_{\sigma_m \tau_2} & \dots & \det g_{\sigma_m \tau_{\binom{n}{k}}}\end{psmallmatrix} \enspace.
  \]
\end{frame}

\begin{frame}[fragile]{Exterior Algebraic Shifting}
  \begin{definition}\label{def:shift}
    The \emph{partial (exterior) shift} of $K$ by $g$ is
    \begin{equation*}
      \Delta^{\FF}_g(K) \coloneqq \bigcup_{k} \Set[\big]{ \sigma \in \tbinom{[n]}{k}; \sigma \text{ indexes a column with a pivot of  } g^{\wedge K_k } }.
    \end{equation*}
  \end{definition}
  \pause
  (Bj\"orner-Kalai '88) For $\EE$ \emph{sufficiently large} we can find $g \in \GL(n, \EE)$ such that $\Delta^{\FF}_g(K) = \Delta^{\FF}(K)$.
  e.g. Take $\EE = \FF(x_{ij} 1 \leq i, j \leq n)$ and $g = (x_{ij})_{1\leq i, j\leq n}$
\end{frame}

\begin{frame}[fragile]{LU decomposition}
  \begin{lemma}
    $\Delta_{gb}(K) = \Delta_g(K)$ for any upper triangular $b$ and any simplicial complex $K$.
  \end{lemma}
  \vspace{8mm}

  \textbf{LU decomposition}
    For $g \in \GL(n, \EE)$ with non zero leading principal minors, there exists
    $l$ lower triangular, $u$ upper triangular such that

    \[
      g = lu \enspace.
    \]
    \vspace{3mm}
    \begin{example}
      $K = \{\{1, 2\}, \{2, 3\}, \{4\}\}$, and  $l^{\wedge K_2} $ reduces to
      \[
        \begin{psmallmatrix}
          x_{13} x_{24} + x_{14} x_{23} & x_{12} x_{24} + x_{14}                                                                                             & x_{24}                                                        & x_{12} x_{23} + x_{13}                                                                                            & x_{23}                                                        & 1                                                  \\[3.5pt]
          0                             & \substack{x_{12} x_{23} x_{24} x_{34} + x_{12} x_{24}^{2} \strut \\ + x_{13} x_{24} x_{34} + x_{14} x_{24} \strut} & \substack{x_{23} x_{24} x_{34} \strut \\ + x_{24}^{2} \strut} & \substack{x_{12} x_{23}^{2} x_{34} + x_{12} x_{23} x_{24} \strut \\ + x_{13} x_{23} x_{34} + x_{14}x_{23} \strut} & \substack{x_{23}^{2} x_{34} \strut \\ + x_{23} x_{24} \strut} & \substack{x_{23} x_{34} \strut \\ + x_{24} \strut}
        \end{psmallmatrix},
      \]
    \end{example}
\end{frame}
