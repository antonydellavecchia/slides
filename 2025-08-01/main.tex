\begin{frame}[fragile]{Algebraic Shifting}
  \begin{definition}
    A \emph{$k$-uniform hypergraph} is a collection $S \subseteq \binom{[n]}{k}$.

    $S$ is \emph{shifted} if for any $\sigma \in S$,
    $\RepVert{\sigma}{i}{j} \in S$ for $i \in \sigma$ and $ j < i$.
  \end{definition}

\begin{frame}[fragile]{Exterior Algebraic Shift}
  \begin{definition}
    Let $\EE \supset \FF$ be a field extension and let $g \in \GL(n, \EE)$.
    \begin{itemize}
      \item The $k$th \emph{compound matrix} $g^{\wedge k}$ is the $\binom{n}{k} \times \binom{n}{k}$-matrix
        of $k$-minors of $g$.
        \item $g^{\wedge S}$ is the $\abs{S} \times \binom{n}{k}$-submatrix of $g^{\wedge k}$ with rows indexed by $S$.
    \end{itemize}
  \end{definition}
  \vspace{8mm}
  Let $\sigma_i \in S$, $\abs{S} = m$ and $\tau_i \in \binom{[n]}{k}$ such that
    \vspace{4mm}
    \[
      g^{\wedge S} \ = \ \begin{psmallmatrix} \det g_{\sigma_1 \tau_1} & \det g_{\sigma_1 \tau_2} & \dots &  \det g_{\sigma_m \tau_{\binom{n}{k}}} \\
        \vdots & \vdots & \vdots & \vdots \\
        \det g_{\sigma_m \tau_1} & \det g_{\sigma_m \tau_2} & \dots & \det g_{\sigma_m \tau_{\binom{n}{k}}}\end{psmallmatrix} \enspace.
    \]

    \pause
The (exterior) \emph{algbraic shift} is,
      \begin{equation*}
        \Delta^{\FF}_g(S) \coloneqq \Set[\big]{ \sigma \in \tbinom{[n]}{k}; \sigma \text{ indexes a column with a pivot of  } g^{\wedge S } }.
      \end{equation*}
\end{frame}

\begin{frame}[fragile]{LU decomposition}
  \begin{lemma}
    $\Delta_{gb}(S) = \Delta_g(S)$ for any upper triangular $b$ and any simplicial complex $S$.
  \end{lemma}
  \vspace{8mm}

  \textbf{LU decomposition}
    For $g \in \GL(n, \EE)$ with non zero leading principal minors, there exists
    $\ell$ lower triangular, $u$ upper triangular such that

    \[
      g = \ell u \enspace.
    \]
    \vspace{3mm}
    \pause
    \begin{example}
      $K = \{\{1, 2\}, \{2, 3\}, \{4\}\}$, and  $\ell^{\wedge S} $ reduces to
      \[
        \begin{psmallmatrix}
          x_{13} x_{24} + x_{14} x_{23} & x_{12} x_{24} + x_{14}                                                                                             & x_{24}                                                        & x_{12} x_{23} + x_{13}                                                                                            & x_{23}                                                        & 1                                                  \\[3.5pt]
          0                             & \substack{x_{12} x_{23} x_{24} x_{34} + x_{12} x_{24}^{2} \strut \\ + x_{13} x_{24} x_{34} + x_{14} x_{24} \strut} & \substack{x_{23} x_{24} x_{34} \strut \\ + x_{24}^{2} \strut} & \substack{x_{12} x_{23}^{2} x_{34} + x_{12} x_{23} x_{24} \strut \\ + x_{13} x_{23} x_{34} + x_{14}x_{23} \strut} & \substack{x_{23}^{2} x_{34} \strut \\ + x_{23} x_{24} \strut} & \substack{x_{23} x_{34} \strut \\ + x_{24} \strut}
        \end{psmallmatrix},
      \]
    \end{example}
\end{frame}

\begin{frame}[fragile]{Monte Carlo Algorithm}
  \begin{itemize}
  \item Randomized algorithm.
  \item Current Standard for computing $\Delta^{\FF}(S)$.
  \item Pick $u$ by sampling from $\FF$.
  \item Computing $\Delta_u^{\FF}(S)$ is fast.
  \item Successful with high probability in characteristic $0$.
  \item Not the case for small fields.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Verification Algorithm}
   \begin{algorithm}[H]
    \begin{algorithmic}[1]
      \REQUIRE $u \in \EE^{n \times n}$, $S$ a $k$-uniform hypergraph.
      \ENSURE \texttt{true} if $\Delta^{\FF}(S) = \Delta^{\FF}_u(S)$; otherwise \texttt{false}
      \STATE $S' \gets \Delta_u(S)$
      \STATE $\sigma_{\text{max}} \gets \lex \max S'$
      \STATE $T \gets \Set{\tau \in \binom{[n]}{k} ; \tau \lex< \sigma_\text{max}}$
      \IF{$\abs{S'} = \abs{T} + 1$}
      \RETURN \texttt{true}
      \ENDIF
      \STATE $m \gets \text{ row echelon form of } \ell^{\wedge S}_{*T}$
      \IF{columns of pivots of $m = S' \setminus \{  \sigma_\text{max}\}$ }
      \RETURN \texttt{true}
      \ENDIF
      \RETURN \texttt{false}
    \end{algorithmic}
    \caption{Verify $u$ computes the algebraic shift of $S$}
    \label{alg:seq}
  \end{algorithm}
  \begin{center}
    $\ell^{\wedge S}$ = \begin{pmatrix}
      |   & |     & \dots & |           & |                    & \dots\\
      c_1 & c_2   & \dots & c_{\abs{T}} &  \sigma_{\text{max}} & \dots\\
      |   & |     & \dots & |           & |                    & \dots 
    \end{pmatrix}
  \end{center}
\end{frame}

\begin{frame}[fragile]{Simplicial Complexes}
  \begin{definition}
    An \emph{abstract simplicial complex} $\Sigma$, is a collection of non empty sets of $[n] := \{1, 2,\ldots, n\}$,
      such that for any $\tau, \sigma \subset [n]$, if $\tau \subset  \sigma \in K$ then $\tau \in \Sigma$.
      \vspace{8mm}
      change lower case k
    \begin{itemize}
    \item $\sigma \in \Sigma$ a \emph{$k$-simplex} or a \emph{face}, dim$(\sigma) = k + 1$.
    \item \emph{$f$-vector} number of $k$-faces for each $k$.
    \end{itemize}
  \end{definition}

\begin{frame}[fragile]{Simplicial Homology}
  \begin{definition}
    Given a field $\FF$, and a simplicial complex $\Sigma$ we define the
    $k$th simplicial homology to be the vector space
    \[
      H_k^{\FF}(\Sigma) \coloneq {k \text{-cycles}}\Big/{ k + 1 \text{-boundaries} }.
    \]
    The $k$th \emph{Betti number} of $\Sigma$ is the dimension of $H_k^{\FF}(\Sigma)$.
  \end{definition}
