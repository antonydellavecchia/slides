%% Slide 1 --- Motivation
\begin{frame}{Serializing Algebraic Data}
  \textbf{Serialization:} translate in-memory data to a form that can be
  \emph{stored} or \emph{transmitted}, then reconstructed on another process
  or machine.
  \bigskip
  \bigskip

  Say we want to store:
  \begin{align*}
    p = 2y^3z^4 + (\mathbf{a} + 3)z^2 + 5\mathbf{a}y + 1
  \end{align*}
  \pause
  \begin{itemize}
  \item Is $y$ considered a coefficient of $z$?
  \item What is $\mathbf{a}$? \pause
  \item How can we guarantee the objects behave as expected on load? \pause
  \item Elements sharing a parent should have the same parent on load.
  \end{itemize}
\end{frame}

%% Slide 2 --- mrdi file format: tree structure + implementation choices
\begin{frame}[fragile]{The \mrdi File Format}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \centering
      \includegraphics[width=\textwidth]{images/tree-coloured}
    \end{column}
    \begin{column}{0.58\textwidth}
      \begin{itemize}
      \item Based on JSON --- a tree, not flat key-value pairs.
      \item \texttt{\_type} carries a \texttt{name} and \texttt{params}.
      \item \texttt{\_ns} pins semantics to a software name + version.
      \item Shared parents (e.g.\ the ring) stored once in \texttt{\_refs},
        addressed by UUID.
      \item Every \OSCAR type implements \texttt{save\_object} /
        \texttt{load\_object} via multiple dispatch.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

%% Slide 4 --- Serializers: multiple dispatch on encoding
\begin{frame}[fragile]{Serializers}
  \begin{itemize}\itemsep=\smallskipamount
  \item Selection of different serializers lets users alter subtrees.
  \item Parametrized \texttt{SerializerState\{S\}} --- \texttt{S} names the
    serializer in use.
  \item Multiple dispatch on \texttt{S} selects the encoding per context ---
    same object, different bytes on disk vs.\ over a socket.
  \end{itemize}
  \bigskip
  \inputminted{jl}{oscar-snippet}
  \footnotesize
\end{frame}

\begin{frame}{Multiprocessing}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \begin{center}
        \workerpoolgraph[0.55]
      \end{center}
    \end{column}
    \begin{column}{0.58\textwidth}
      \textbf{\texttt{OscarWorkerPool}} --- subtype of \Julia's
      \texttt{AbstractWorkerPool}, built on the serializers above.
      \begin{itemize}
      \item Sends context to workers once, then reuses an
        \texttt{IPCSerializer} for every message.
      \item \Julia's \texttt{remotecall} and \texttt{pmap} work directly with
        \OSCAR types.
      \item Works with cluster managers
        (\texttt{SlurmClusterManager.jl}) --- 1000+ cpus.
      \item \textbf{32 processes, \SI{150}{\giga\byte}: $<3$ hours vs.\
        $>24$ hours (1 process)} on a degree-5 vanishing-ideal computation.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

%% Slide 6 --- Implementation details: type_and_params / load_object
\begin{frame}[fragile]{Under the Hood: \texttt{type\_and\_params}}
  \texttt{type\_and\_params(obj)} returns the context needed to rebuild
  \texttt{obj} --- e.g.\ its parent ring --- kept out of \texttt{data} and
  stored once as a shared \texttt{\_refs} entry.
  \bigskip

  \inputminted{jl}{type-and-params-snippet}
  \footnotesize
  \begin{itemize}
  \item \texttt{load\_object} dispatches on
    \texttt{TypeAndParams\{Type, ParamType\}} --- the whole type tree is
    resolved \emph{before} any \texttt{data} is touched.
  \item Rule: never call \texttt{save\_typed\_object} inside
    \texttt{save\_object} --- keeps type and data subtrees separate, so
    files stay easy to validate and upgrade.
  \end{itemize}
